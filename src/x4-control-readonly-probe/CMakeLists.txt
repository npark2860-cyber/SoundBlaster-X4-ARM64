cmake_minimum_required(VERSION 3.25)
project(SoundBlasterX4ReadOnlyCapabilityProbe LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This diagnostic targets Windows only.")
endif()
add_executable(x4-control-readonly-probe main.cpp)
target_compile_features(x4-control-readonly-probe PRIVATE cxx_std_17)
target_compile_definitions(x4-control-readonly-probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
if(MSVC)
    target_compile_options(x4-control-readonly-probe PRIVATE /W4 /EHsc)
    set_property(TARGET x4-control-readonly-probe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
target_link_libraries(x4-control-readonly-probe PRIVATE setupapi.lib advapi32.lib)
