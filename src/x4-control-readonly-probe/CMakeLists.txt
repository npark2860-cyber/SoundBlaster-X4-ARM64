cmake_minimum_required(VERSION 3.25)
project(SoundBlasterX4ReadOnlyCapabilityProbe LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This diagnostic targets Windows only.")
endif()

function(configure_probe target source)
    add_executable(${target} ${source})
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /EHsc)
        set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    target_link_libraries(${target} PRIVATE setupapi.lib advapi32.lib)
endfunction()

configure_probe(x4-control-readonly-probe main.cpp)
configure_probe(x4-ctcdc-challenge-capture challenge-capture.cpp)
